 match not access-group name ACL-SAMPLE-Q

  set qos-group 10
  set cos 5

  police cir percent 20 conform-action transmit exceed-action set-dscp-transmit af11 violate-action drop

 class cm-m-voice
  priority percent 20

  bandwidth remaining percent 40
  shape peak 4000000
  random-detect dscp-based
  random-detect dscp 34 24 40

  shape average percent 30
  queue-limit 64000 bytes
 class class-default
  police cir 2000000 bc 8000 pir 4000000 be 16000
   conform-action transmit
   exceed-action transmit
   violate-action drop
  random-detect
