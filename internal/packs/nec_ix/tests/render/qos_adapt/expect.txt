  match ip access-list qos-d1

  match ip access-list qos-d2

    priority
  class cm-q-video
    bandwidth percent 60
  class class-local
  class class-default
    bandwidth percent 35

  service-policy enable
  service-policy output pm-q-out
  traffic-shape rate 10000000
  no shutdown
